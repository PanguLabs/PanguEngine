cmake_minimum_required(VERSION 4.0)
project(PanguAbiBuildContract LANGUAGES CXX)
if(NOT PANGU_ABI_SOURCE_ROOT)
    message(FATAL_ERROR "PANGU_ABI_SOURCE_ROOT is required")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include("${PANGU_ABI_SOURCE_ROOT}/cmake/Platform/Common/BuildOptions.cmake")
include("${PANGU_ABI_SOURCE_ROOT}/cmake/Platform/Common/Module.cmake")
include("${PANGU_ABI_SOURCE_ROOT}/cmake/AbiProfile/AbiProfile.cmake")

configure_file(Base.cpp "${CMAKE_CURRENT_BINARY_DIR}/Base.cpp" COPYONLY)
add_library(FixtureBase STATIC "${CMAKE_CURRENT_BINARY_DIR}/Base.cpp")
add_library(FixtureModule STATIC Module.cpp)
add_library(TechA STATIC TechA.cpp)
add_library(TechB STATIC TechB.cpp)
target_link_libraries(TechA PRIVATE PanguBuildOptions)
target_link_libraries(TechB PRIVATE PanguBuildOptions)
set_target_properties(TechA TechB PROPERTIES CXX_STANDARD 23 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
pangu_configure_module(FixtureBase)
pangu_configure_module(FixtureModule)
if(PANGU_ABI_INTERFACE_CYCLE)
    add_library(FixtureCycleFacade INTERFACE)
    target_link_libraries(FixtureCycleFacade INTERFACE TechB)
    target_link_libraries(TechA PRIVATE FixtureCycleFacade)
else()
    target_link_libraries(TechA PRIVATE TechB)
endif()
target_link_libraries(TechB PRIVATE TechA)
if(PANGU_ABI_TRANSITIVE_TECH)
    # A CMake INTERFACE dependency carries the technical SCC transitively.
    # The managed Module has no direct LINK_LIBRARIES entry for TechA.
    add_library(FixtureTechnicalInterface INTERFACE)
    if(PANGU_ABI_CONDITIONAL_TECH)
        target_link_libraries(FixtureTechnicalInterface INTERFACE "$<$<CONFIG:Release>:TechA>")
    else()
        target_link_libraries(FixtureTechnicalInterface INTERFACE TechA)
    endif()
    target_link_libraries(FixtureModule PRIVATE FixtureBase FixtureTechnicalInterface)
else()
    target_link_libraries(FixtureModule PRIVATE FixtureBase TechA)
endif()

pangu_register_abi_module(FixtureBase)
pangu_register_abi_module(FixtureModule)
pangu_register_abi_static(TechA)
pangu_register_abi_static(TechB)
set(_graphTargets FixtureBase FixtureModule TechA TechB PanguBuildOptions)
if(PANGU_ABI_INTERFACE_CYCLE)
    list(APPEND _graphTargets FixtureCycleFacade)
endif()
if(PANGU_ABI_TRANSITIVE_TECH)
    list(APPEND _graphTargets FixtureTechnicalInterface)
endif()
pangu_finalize_abi_profiles(TARGETS ${_graphTargets})
