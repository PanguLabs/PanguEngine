cmake_minimum_required(VERSION 4.0)
project(PanguAbiLauncherContract LANGUAGES CXX)
if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang" OR
   CMAKE_CXX_COMPILER_VERSION VERSION_LESS 23.0 OR
   NOT CMAKE_CXX_COMPILER_VERSION VERSION_LESS 24.0)
    message(FATAL_ERROR "Launcher contract fixtures require native Clang 23.x")
endif()

set(CMAKE_CXX_STANDARD 23)
include("${PANGU_ABI_SOURCE_ROOT}/cmake/Platform/Common/BuildOptions.cmake")
include("${PANGU_ABI_SOURCE_ROOT}/cmake/Platform/Common/Module.cmake")
include("${PANGU_ABI_SOURCE_ROOT}/cmake/AbiProfile/AbiProfile.cmake")
set(_original "${CMAKE_COMMAND};-E;env")
if(PANGU_LAUNCHER_MODE STREQUAL "global")
    set(CMAKE_CXX_COMPILER_LAUNCHER "${_original}")
endif()
add_library(LauncherFixture STATIC "${PANGU_ABI_SOURCE_ROOT}/Tests/AbiProfile/Fixtures/BuildContract/Base.cpp")
target_link_libraries(LauncherFixture PRIVATE PanguBuildOptions)
if(PANGU_LAUNCHER_MODE STREQUAL "target")
    set_property(TARGET LauncherFixture PROPERTY CXX_COMPILER_LAUNCHER "${_original}")
endif()
if(PANGU_LAUNCHER_KIND STREQUAL "module")
    pangu_configure_module(LauncherFixture)
    pangu_register_abi_module(LauncherFixture)
else()
    pangu_register_abi_static(LauncherFixture)
endif()
get_target_property(_after LauncherFixture CXX_COMPILER_LAUNCHER)
file(WRITE "${CMAKE_BINARY_DIR}/launcher-after.txt" "${_after}")
