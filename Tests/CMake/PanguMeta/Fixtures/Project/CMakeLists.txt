cmake_minimum_required(VERSION 4.0)
include("${PANGU_ENGINE_SOURCE_DIR}/cmake/ValidateBuildEnvironment.cmake")
project(PanguLayoutProject LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/Bin")
if(EXISTS "${PANGU_ENGINE_SOURCE_DIR}/cmake/PanguEngine.cmake")
    include("${PANGU_ENGINE_SOURCE_DIR}/cmake/PanguEngine.cmake")
    pangu_add_engine(SOURCE_DIR "${PANGU_ENGINE_SOURCE_DIR}" ENGINE_BUILD_ROOT "${PANGU_ENGINE_BUILD_ROOT}")
else()
    # 保留旧接线以证明失败来自真实产物归属，而不是缺失命令。
    add_subdirectory("${PANGU_ENGINE_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/Engine")
endif()
include("${PANGU_ENGINE_SOURCE_DIR}/cmake/PanguPlugins.cmake")
pangu_add_enabled_plugins(ENGINE_ROOT "${PANGU_FIXTURE_ENGINE_ROOT}"
    PROJECT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}" ENABLED pangu.layout-value project.layout-value)
add_executable(LayoutHost Main.cpp)
add_dependencies(LayoutHost PanguLayoutOwnedProbe)
target_link_libraries(LayoutHost PRIVATE PanguBuildOptions PanguLayoutValue PanguProjectValue Pangu::Module)
if(WIN32 AND PANGU_LAYOUT_CONFIG_HOST_DIR)
    set(configDirectory "${CMAKE_BINARY_DIR}/Bin/$<CONFIG>")
    if(PANGU_LAYOUT_CONFIG_HOST_DIR STREQUAL "Unsupported")
        set(configDirectory "$<TARGET_PROPERTY:LayoutHost,RUNTIME_OUTPUT_DIRECTORY>/Debug")
    endif()
    set_target_properties(LayoutHost PROPERTIES RUNTIME_OUTPUT_DIRECTORY_DEBUG "${configDirectory}")
    add_executable(LayoutOtherHost Main.cpp)
    target_link_libraries(LayoutOtherHost PRIVATE PanguBuildOptions PanguLayoutValue PanguProjectValue Pangu::Module)
    set_target_properties(LayoutOtherHost PROPERTIES
        RUNTIME_OUTPUT_DIRECTORY_DEBUG "${CMAKE_BINARY_DIR}/Bin/OtherDebug")
    pangu_stage_host_runtime(TARGET LayoutOtherHost)
    file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/OtherHost.txt" CONTENT "$<TARGET_FILE:LayoutOtherHost>")
endif()
if(COMMAND pangu_stage_host_runtime)
    pangu_stage_host_runtime(TARGET LayoutHost)
endif()
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/Layout.txt" CONTENT
    "engine_binary=$<TARGET_PROPERTY:PanguFoundation,BINARY_DIR>\nengine_file=$<TARGET_FILE:PanguFoundation>\nengine_export=$<TARGET_PROPERTY:PanguFoundation,INTERFACE_INCLUDE_DIRECTORIES>\nengine_owner=$<TARGET_PROPERTY:PanguFoundation,PANGU_SOURCE_OWNER>\nplugin_binary=$<TARGET_PROPERTY:PanguLayoutValue,BINARY_DIR>\nplugin_file=$<TARGET_FILE:PanguLayoutValue>\nplugin_owner=$<TARGET_PROPERTY:PanguLayoutValue,PANGU_SOURCE_OWNER>\nowned_file=$<TARGET_FILE:PanguLayoutOwnedProbe>\nproject_binary=$<TARGET_PROPERTY:PanguProjectValue,BINARY_DIR>\nproject_file=$<TARGET_FILE:PanguProjectValue>\nproject_owner=$<TARGET_PROPERTY:PanguProjectValue,PANGU_SOURCE_OWNER>\nhost=$<TARGET_FILE:LayoutHost>\n")
