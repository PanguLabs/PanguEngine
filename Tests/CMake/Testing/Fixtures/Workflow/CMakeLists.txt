cmake_minimum_required(VERSION 4.0)
project(PanguWorkflowFixture LANGUAGES CXX)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
    message(FATAL_ERROR "Workflow fixture requires Clang")
endif()
include(CTest)
include("${PANGU_ENGINE_ROOT}/cmake/PanguTesting.cmake")
option(PANGU_BUILD_ENGINE_TESTS "Enable fixture Engine tests" ON)
if(NOT FIXTURE_OWNER)
    set(FIXTURE_OWNER engine)
endif()
pangu_require_full_testing(OWNER "${FIXTURE_OWNER}")
pangu_test_context(OWNER "${FIXTURE_OWNER}" ENABLED ON)
pangu_tests_enabled(enabled)
if(NOT enabled)
    return()
endif()
include("${PANGU_ENGINE_ROOT}/cmake/PanguTestDependencies.cmake")
pangu_require_googletest(RTTI ON)
foreach(tier IN ITEMS Fast Full)
    add_executable("Fixture${tier}" "Sample${tier}.cpp")
    target_compile_features("Fixture${tier}" PRIVATE cxx_std_23)
    target_compile_options("Fixture${tier}" PRIVATE -fno-exceptions)
    target_link_libraries("Fixture${tier}" PRIVATE ThirdParty::GoogleTestMain)
    string(TOLOWER "${tier}" level)
    pangu_register_gtest(TARGET "Fixture${tier}" BOUNDARY unit AREA testing TIER "${level}")
endforeach()
